# Query Overpass per cercare le strade a Palermo
def get_street_geometry(street_name):
    """Ottiene la geometria di una strada da OSM"""
    overpass_url = "http://overpass-api.de/api/interpreter"
    
    # Rimuovi ", Palermo" dal nome per la query
    clean_name = street_name.replace(", Palermo", "")
    
    overpass_query = f"""
    [out:json];
    area["name"="Palermo"]["admin_level"="8"]->.searchArea;
    (
      way["name"="{clean_name}"](area.searchArea);
      relation["name"="{clean_name}"](area.searchArea);
    );
    out geom;
    """
    
    try:
        response = requests.get(overpass_url, params={'data': overpass_query}, timeout=30)
        data = response.json()
        return data.get('elements', [])
    except Exception as e:
        print(f"Errore per {street_name}: {e}")
        return []

# Raccogli tutte le geometrie
print("Recupero dati da OpenStreetMap...")
all_features = []

for i, strada in enumerate(strade_chiuse):
    print(f"[{i+1}/{len(strade_chiuse)}] Cerco: {strada}")
    elements = get_street_geometry(strada)
    
    for element in elements:
        if element['type'] == 'way' and 'geometry' in element:
            coords = [[node['lat'], node['lon']] for node in element['geometry']]
            all_features.append({
                'name': strada,
                'coords': coords,
                'type': 'way'
            })
    
    # Pausa per non sovraccaricare l'API
    time.sleep(1)

print(f"\nTrovate {len(all_features)} geometrie")

# Crea la mappa
